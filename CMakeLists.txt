cmake_minimum_required(VERSION 3.20)
project(VIC_GigE VERSION 0.4.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "VICGigE currently requires Windows/Winsock2")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT OpenCV_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/out/g03/opencv-sdk/opencv/build/OpenCVConfig.cmake")
  set(OpenCV_DIR "${CMAKE_CURRENT_SOURCE_DIR}/out/g03/opencv-sdk/opencv/build")
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
add_library(vic_core src/core/camera.cpp)
target_include_directories(vic_core PUBLIC src PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vic_core PUBLIC ${OpenCV_LIBS})
add_library(vic_gvcp src/gige/gvcp.cpp src/gige/gvsp.cpp)
target_link_libraries(vic_gvcp PUBLIC vic_core)
target_include_directories(vic_gvcp PUBLIC src)
target_compile_definitions(vic_gvcp PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_executable(VICGigE src/gige/main.cpp src/gige/stream_sender.cpp)
add_executable(VICEngine src/control06/main.cpp)
target_include_directories(VICEngine PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(VICEngine PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(VICEngine PRIVATE vic_core)
if(MSVC)
  target_compile_options(VICEngine PRIVATE /W4 /permissive- /utf-8)
endif()
target_link_libraries(VICGigE PRIVATE vic_gvcp vic_core ws2_32 iphlpapi)
add_executable(VICGigEControl poc/tap-camera/main.cpp poc/tap-camera/engine.cpp poc/tap-camera/offline.cpp)
target_include_directories(VICGigEControl PRIVATE poc/tap-camera/third_party)
target_compile_definitions(VICGigEControl PRIVATE VIC_G05_CONTROL)
target_link_libraries(VICGigEControl PRIVATE vic_gvcp vic_core ws2_32 iphlpapi ole32)
add_executable(VICGigE06Control poc/tap-camera/main.cpp poc/tap-camera/engine.cpp poc/tap-camera/offline.cpp)
target_include_directories(VICGigE06Control PRIVATE poc/tap-camera/third_party ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(VICGigE06Control PRIVATE VIC_G05_CONTROL VIC_06_SOURCES)
target_link_libraries(VICGigE06Control PRIVATE vic_gvcp vic_core ws2_32 iphlpapi ole32)
set_target_properties(VICGigE06Control PROPERTIES OUTPUT_NAME VICGigEControl RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/gige06")
if(MSVC)
  target_compile_options(vic_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(vic_gvcp PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(VICGigE PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(VICGigEControl PRIVATE /W4 /permissive- /utf-8)
endif()
foreach(runtime opencv_world4120.dll opencv_world4120d.dll opencv_videoio_ffmpeg4120_64.dll)
  configure_file("${OpenCV_DIR}/x64/vc16/bin/${runtime}" "${CMAKE_CURRENT_BINARY_DIR}/${runtime}" COPYONLY)
  configure_file("${OpenCV_DIR}/x64/vc16/bin/${runtime}" "${CMAKE_CURRENT_BINARY_DIR}/gige06/${runtime}" COPYONLY)
endforeach()

  target_include_directories(control06_unit PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(control06_unit PRIVATE vic_core)
  add_test(NAME control06_unit COMMAND control06_unit)

  target_include_directories(g05_control PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(g05_control PRIVATE vic_gvcp)
  add_test(NAME g05_control COMMAND g05_control)
  add_executable(g04_compat tests/automated/g04_compat.cpp)
  target_link_libraries(g04_compat PRIVATE vic_gvcp)
  add_test(NAME g04_compat COMMAND g04_compat)
  add_executable(g03_core tests/automated/g03_core.cpp)
  target_include_directories(g03_core PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(g03_core PRIVATE vic_core vic_gvcp)
  add_test(NAME g03_core COMMAND g03_core)

  add_test(NAME g01_unit COMMAND g01_unit)
