cmake_minimum_required(VERSION 3.20)
project(VIC_TAP_POC VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(VIC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(OpenCV_DIR "${VIC_ROOT}/out/g03/opencv-sdk/opencv/build")
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
add_executable(VICTapPoC main.cpp engine.cpp offline.cpp
  "${VIC_ROOT}/src/gige/gvcp.cpp" "${VIC_ROOT}/src/gige/gvsp.cpp" "${VIC_ROOT}/src/core/camera.cpp")
target_include_directories(VICTapPoC PRIVATE "${VIC_ROOT}/src" ${OpenCV_INCLUDE_DIRS} third_party)
target_compile_definitions(VICTapPoC PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(VICTapPoC PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(VICTapPoC PRIVATE ${OpenCV_LIBS} ws2_32 iphlpapi ole32)
foreach(runtime opencv_world4120.dll opencv_videoio_ffmpeg4120_64.dll)
  configure_file("${OpenCV_DIR}/x64/vc16/bin/${runtime}" "${CMAKE_CURRENT_BINARY_DIR}/${runtime}" COPYONLY)
endforeach()
enable_testing()
add_test(NAME tap_offline COMMAND VICTapPoC --offline "${CMAKE_CURRENT_BINARY_DIR}/offline-test")
